/**
 * Procedural track generation.
 *
 * The track is represented as a continuous parametric curve in world-space.
 * At any distance `t` along the track we can query:
 *   - horizontal offset (xOffset)
 *   - vertical offset (yOffset)
 *   - horizontal curvature derivative (dX) — used for banking/lean
 *   - vertical curvature derivative (dY)
 *
 * Biome configs drive the amplitude and frequency of these curves.
 * Multiple biomes can be blended by lerping their configs at transition zones.
 */

import { fbm1D } from './noise';
import type { BiomeConfig } from '../biomes/types';

export interface TrackSample {
  xOffset: number;  // lateral displacement from center, in NDC-like units
  yOffset: number;  // vertical displacement
  dX: number;       // rate of change of xOffset (curvature)
  dY: number;       // rate of change of yOffset
}

const DX_STEP = 0.1; // finite-difference step for derivative

export function sampleTrack(t: number, biome: BiomeConfig): TrackSample {
  const seed = biome.seedOffset;

  const xAt = (s: number) =>
    fbm1D(s * biome.curvatureFrequency, seed + 1, 4) * biome.curvatureAmplitude;

  const yAt = (s: number) =>
    fbm1D(s * biome.verticalFrequency, seed + 2, 3) * biome.verticalAmplitude;

  const xOffset = xAt(t);
  const yOffset = yAt(t);
  const dX = (xAt(t + DX_STEP) - xAt(t - DX_STEP)) / (2 * DX_STEP);
  const dY = (yAt(t + DX_STEP) - yAt(t - DX_STEP)) / (2 * DX_STEP);

  return { xOffset, yOffset, dX, dY };
}
