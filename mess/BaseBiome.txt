import type { BiomeConfig, MineObject, MineObjectType } from '@/types';
import { createRNG } from '@/utils/math';

/**
 * Base class for biome generation
 * Extend this to create new biome types
 */
export abstract class BaseBiome {
  protected config: BiomeConfig;
  protected rng: ReturnType<typeof createRNG>;
  
  constructor(config: BiomeConfig, seed: number) {
    this.config = config;
    this.rng = createRNG(seed);
  }
  
  getConfig(): BiomeConfig {
    return this.config;
  }
  
  getId(): string {
    return this.config.id;
  }
  
  getName(): string {
    return this.config.name;
  }
  
  /**
   * Generate objects for a chunk at the given z position
   */
  abstract generateObjects(zStart: number, zEnd: number, chunkId: number): MineObject[];
  
  /**
   * Get a weighted random object type
   */
  protected getRandomObjectType(): MineObjectType | null {
    const weights = this.config.objectWeights;
    const entries = Object.entries(weights) as [MineObjectType, number][];
    
    if (entries.length === 0) return null;
    
    const totalWeight = entries.reduce((sum, [, weight]) => sum + weight, 0);
    let random = this.rng.next() * totalWeight;
    
    for (const [type, weight] of entries) {
      random -= weight;
      if (random <= 0) return type;
    }
    
    return entries[0][0];
  }
  
  /**
   * Check if an object should spawn based on frequency
   */
  protected shouldSpawn(frequency: number): boolean {
    return this.rng.chance(frequency);
  }
}
