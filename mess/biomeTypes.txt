/**
 * BiomeConfig defines all visual and generative parameters for a single biome.
 * Adding a new biome is as simple as creating a new object conforming to this interface.
 */
export interface BiomeConfig {
  id: string;
  name: string;

  // Tunnel geometry
  tunnelWidth: number;       // base half-width of the tunnel at the horizon (world units)
  tunnelHeight: number;      // base half-height of the tunnel at the horizon (world units)
  tunnelRoughness: number;   // 0–1 how jagged the walls are

  // Track curvature
  curvatureAmplitude: number;  // max horizontal deviation per chunk
  curvatureFrequency: number;  // how frequently curves change (lower = longer sweeping curves)
  verticalAmplitude: number;   // max vertical deviation per chunk
  verticalFrequency: number;

  // Colors (CSS strings or hex)
  colors: {
    fog: string;           // distant haze color
    wallNear: string;      // wall color close to camera
    wallFar: string;       // wall color at distance (lerps toward fog)
    ceiling: string;
    floor: string;
    trackRail: string;
    trackTie: string;
    glowPrimary: string;   // torch/lamp glow color
    glowSecondary: string;
    ambientLight: string;  // overall scene tint
  };

  // Atmosphere
  fogDensity: number;        // 0–1, how quickly geometry fades to fog
  dustParticles: boolean;
  dripDrops: boolean;

  // Lighting
  torchSpacing: number;      // world units between torches
  torchFlicker: boolean;

  // Generation seed offset (lets biomes feel distinct even with the same base noise)
  seedOffset: number;
}
