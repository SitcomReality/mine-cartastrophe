/**
 * useGameLoop
 *
 * Drives the render loop via requestAnimationFrame.
 * Accepts a callback that receives the current timestamp.
 * Automatically cancels the loop on unmount.
 */
import { useEffect, useRef } from 'react';

export function useGameLoop(tick: (timestamp: number) => void) {
  const tickRef = useRef(tick);
  tickRef.current = tick; // always call the latest version

  useEffect(() => {
    let rafId = 0;
    let running = true;

    const loop = (ts: number) => {
      if (!running) return;
      tickRef.current(ts);
      rafId = requestAnimationFrame(loop);
    };

    rafId = requestAnimationFrame(loop);

    return () => {
      running = false;
      cancelAnimationFrame(rafId);
    };
  }, []);
}
