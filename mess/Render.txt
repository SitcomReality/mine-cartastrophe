import type { Chunk, MineObject, BiomeConfig, Color } from '@/types';
import type { Camera } from './Camera';
import { color, clamp } from '@/utils/math';
import { RAIL_SPACING, FAR_PLANE, AMBIENT_FLICKER_SPEED, AMBIENT_FLICKER_AMOUNT } from '@/constants';

interface RenderContext {
  ctx: CanvasRenderingContext2D;
  width: number;
  height: number;
  camera: Camera;
  biomeConfig: BiomeConfig;
  time: number;
}

/**
 * High-performance canvas renderer for the mine
 * Uses painter's algorithm (back-to-front) for proper depth
 */
export class Renderer {
  private canvas: HTMLCanvasElement;
  private ctx: CanvasRenderingContext2D;
  private width = 800;
  private height = 600;
  
  // Stats tracking
  private drawCalls = 0;
  private objectsRendered = 0;
  
  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d', { 
      alpha: false,
      desynchronized: true // Performance hint
    })!;
    
    this.resize();
  }
  
  resize(): void {
    const dpr = Math.min(window.devicePixelRatio, 2); // Cap at 2x for performance
    const rect = this.canvas.getBoundingClientRect();
    
    this.width = rect.width;
    this.height = rect.height;
    
    this.canvas.width = this.width * dpr;
    this.canvas.height = this.height * dpr;
    
    this.ctx.scale(dpr, dpr);
  }
  
  /**
   * Main render function
   */
  render(
    camera: Camera,
    chunks: Chunk[],
    biomeConfig: BiomeConfig,
    time: number
  ): void {
    this.drawCalls = 0;
    this.objectsRendered = 0;
    
    const renderCtx: RenderContext = {
      ctx: this.ctx,
      width: this.width,
      height: this.height,
      camera,
      biomeConfig,
      time,
    };
    
    // Clear with fog color (background)
    this.ctx.fillStyle = color.toRgb(biomeConfig.fogColor);
    this.ctx.fillRect(0, 0, this.width, this.height);
    this.drawCalls++;
    
    // Render chunks back to front
    for (const chunk of chunks) {
      this.renderChunk(renderCtx, chunk);
    }
    
    // Apply screen-space effects
    this.renderAtmosphericEffects(renderCtx);
  }
  
  /**
   * Render a single chunk
   */
  private renderChunk(ctx: RenderContext, chunk: Chunk): void {
    // Render tunnel geometry
    this.renderTunnel(ctx, chunk);
    
    // Render track
    this.renderTrack(ctx, chunk);
    
    // Render objects (sorted by depth)
    const sortedObjects = [...chunk.objects].sort(
      (a, b) => b.position.z - a.position.z
    );
    
    for (const obj of sortedObjects) {
      this.renderObject(ctx, obj);
    }
  }
  
  /**
   * Render the tunnel walls, floor, and ceiling
   * Handles both enclosed tunnels and open caverns with abysses
   */
  private renderTunnel(ctx: RenderContext, chunk: Chunk): void {
    const { camera, biomeConfig } = ctx;
    const { tunnelWidth, tunnelHeight, hasLeftWall, hasRightWall, hasCeiling, hasFloor, floorEdgeWidth } = biomeConfig;
    
    const segments = 5;
    const segmentLength = (chunk.zEnd - chunk.zStart) / segments;
    
    // Render abyss void first (behind everything)
    if (!hasFloor || !hasLeftWall || !hasRightWall) {
      this.renderAbyssVoid(ctx, chunk);
    }
    
    for (let i = segments - 1; i >= 0; i--) {
      const zFar = chunk.zStart + (i + 1) * segmentLength;
      const zNear = chunk.zStart + i * segmentLength;
      
      // Calculate fog for this segment
      const depthFar = camera.getDepthFactor(zFar, FAR_PLANE);
      const depthNear = camera.getDepthFactor(zNear, FAR_PLANE);
      const fogNear = this.calculateFog(depthNear, biomeConfig.fogDensity);
      const fogFar = this.calculateFog(depthFar, biomeConfig.fogDensity);
      
      // Calculate light flickering
      const flicker = 1 - Math.sin(ctx.time * AMBIENT_FLICKER_SPEED) * AMBIENT_FLICKER_AMOUNT;
      
      // Project corner points based on geometry type
      const halfWidth = tunnelWidth / 2;
      const ledgeHalfWidth = floorEdgeWidth / 2;
      
      if (hasFloor) {
        // Full floor
        const nearLeft = camera.project({ x: -halfWidth, y: 0, z: zNear });
        const nearRight = camera.project({ x: halfWidth, y: 0, z: zNear });
        const farLeft = camera.project({ x: -halfWidth, y: 0, z: zFar });
        const farRight = camera.project({ x: halfWidth, y: 0, z: zFar });
        
        if (nearLeft && nearRight && farLeft && farRight) {
          const floorColorNear = this.applyFogAndLight(biomeConfig.floorColor, biomeConfig.fogColor, fogNear, flicker);
          const floorColorFar = this.applyFogAndLight(biomeConfig.floorColor, biomeConfig.fogColor, fogFar, flicker);
          
          this.drawQuadGradient(ctx.ctx, farLeft, farRight, nearRight, nearLeft, floorColorFar, floorColorNear);
        }
      } else {
        // Narrow ledge/bridge - render just the platform
        const nearLeft = camera.project({ x: -ledgeHalfWidth, y: 0, z: zNear });
        const nearRight = camera.project({ x: ledgeHalfWidth, y: 0, z: zNear });
        const farLeft = camera.project({ x: -ledgeHalfWidth, y: 0, z: zFar });
        const farRight = camera.project({ x: ledgeHalfWidth, y: 0, z: zFar });
        
        if (nearLeft && nearRight && farLeft && farRight) {
          const floorColorNear = this.applyFogAndLight(biomeConfig.floorColor, biomeConfig.fogColor, fogNear, flicker);
          const floorColorFar = this.applyFogAndLight(biomeConfig.floorColor, biomeConfig.fogColor, fogFar, flicker);
          
          this.drawQuadGradient(ctx.ctx, farLeft, farRight, nearRight, nearLeft, floorColorFar, floorColorNear);
        }
        
        // Render ledge edge/side (shows depth of the ledge)
        this.renderLedgeEdge(ctx, zNear, zFar, ledgeHalfWidth, hasLeftWall, hasRightWall, fogNear, fogFar, flicker);
      }
      
      // Left wall (only if present)
      if (hasLeftWall) {
        const nearLeft = camera.project({ x: -halfWidth, y: 0, z: zNear });
        const nearTopLeft = camera.project({ x: -halfWidth, y: tunnelHeight, z: zNear });
        const farLeft = camera.project({ x: -halfWidth, y: 0, z: zFar });
        const farTopLeft = camera.project({ x: -halfWidth, y: tunnelHeight, z: zFar });
        
        if (nearLeft && nearTopLeft && farLeft && farTopLeft) {
          const wallColorNear = this.applyFogAndLight(biomeConfig.wallColor, biomeConfig.fogColor, fogNear, flicker);
          const wallColorFar = this.applyFogAndLight(biomeConfig.wallColor, biomeConfig.fogColor, fogFar, flicker);
          
          this.drawQuadGradient(ctx.ctx, farLeft, farTopLeft, nearTopLeft, nearLeft, wallColorFar, wallColorNear);
        }
      }
      
      // Right wall (only if present)
      if (hasRightWall) {
        const nearRight = camera.project({ x: halfWidth, y: 0, z: zNear });
        const nearTopRight = camera.project({ x: halfWidth, y: tunnelHeight, z: zNear });
        const farRight = camera.project({ x: halfWidth, y: 0, z: zFar });
        const farTopRight = camera.project({ x: halfWidth, y: tunnelHeight, z: zFar });
        
        if (nearRight && nearTopRight && farRight && farTopRight) {
          const wallColorNear = this.applyFogAndLight(biomeConfig.wallColor, biomeConfig.fogColor, fogNear, flicker);
          const wallColorFar = this.applyFogAndLight(biomeConfig.wallColor, biomeConfig.fogColor, fogFar, flicker);
          
          this.drawQuadGradient(ctx.ctx, farRight, farTopRight, nearTopRight, nearRight, wallColorFar, wallColorNear);
        }
      }
      
      // Ceiling (only if present)
      if (hasCeiling) {
        const nearTopLeft = camera.project({ x: -halfWidth, y: tunnelHeight, z: zNear });
        const nearTopRight = camera.project({ x: halfWidth, y: tunnelHeight, z: zNear });
        const farTopLeft = camera.project({ x: -halfWidth, y: tunnelHeight, z: zFar });
        const farTopRight = camera.project({ x: halfWidth, y: tunnelHeight, z: zFar });
        
        if (nearTopLeft && nearTopRight && farTopLeft && farTopRight) {
          const ceilingColorNear = this.applyFogAndLight(biomeConfig.ceilingColor, biomeConfig.fogColor, fogNear, flicker);
          const ceilingColorFar = this.applyFogAndLight(biomeConfig.ceilingColor, biomeConfig.fogColor, fogFar, flicker);
          
          this.drawQuadGradient(ctx.ctx, farTopLeft, farTopRight, nearTopRight, nearTopLeft, ceilingColorFar, ceilingColorNear);
        }
      }
      
      // Add subtle rock texture details on walls (only for near segments for performance)
      if (i < 3 && fogNear < 0.7 && (hasLeftWall || hasRightWall)) {
        this.renderWallDetails(ctx, zNear, zFar, halfWidth, tunnelHeight, fogNear);
      }
    }
  }
  
  /**
   * Render the void/abyss below and beside the track
   */
  private renderAbyssVoid(ctx: RenderContext, chunk: Chunk): void {
    const { camera, biomeConfig, time } = ctx;
    const { floorEdgeWidth, abyssGlowColor, abyssGlowIntensity, hasLeftWall, hasRightWall } = biomeConfig;
    
    const ledgeHalfWidth = floorEdgeWidth / 2;
    const voidWidth = 30; // How far the void extends
    const voidDepth = -50; // How deep the void goes visually
    
    // Project void boundaries
    const zNear = chunk.zStart;
    const zFar = chunk.zEnd;
    
    const depthFar = camera.getDepthFactor(zFar, FAR_PLANE);
    const fogFar = this.calculateFog(depthFar, biomeConfig.fogDensity * 0.5);
    
    // Render void on left side
    if (!hasLeftWall) {
      const nearLedge = camera.project({ x: -ledgeHalfWidth, y: 0, z: zNear });
      const farLedge = camera.project({ x: -ledgeHalfWidth, y: 0, z: zFar });
      const nearVoid = camera.project({ x: -ledgeHalfWidth - voidWidth, y: voidDepth, z: zNear });
      const farVoid = camera.project({ x: -ledgeHalfWidth - voidWidth, y: voidDepth, z: zFar });
      
      if (nearLedge && farLedge && nearVoid && farVoid) {
        // Render void gradient (darkness fading into nothingness)
        const voidGradient = ctx.ctx.createLinearGradient(
          nearLedge.x, nearLedge.y,
          nearVoid.x, this.height
        );
        
        voidGradient.addColorStop(0, color.toRgba({ ...biomeConfig.fogColor, a: 0.9 }));
        voidGradient.addColorStop(0.4, color.toRgba({ r: 3, g: 2, b: 5, a: 1 }));
        voidGradient.addColorStop(1, color.toRgba({ r: 0, g: 0, b: 0, a: 1 }));
        
        ctx.ctx.fillStyle = voidGradient;
        ctx.ctx.beginPath();
        ctx.ctx.moveTo(nearLedge.x, nearLedge.y);
        ctx.ctx.lineTo(farLedge.x, farLedge.y);
        ctx.ctx.lineTo(0, this.height);
        ctx.ctx.lineTo(0, nearVoid.y);
        ctx.ctx.closePath();
        ctx.ctx.fill();
        this.drawCalls++;
      }
      
      // Abyss glow effect from below
      if (abyssGlowColor && abyssGlowIntensity > 0) {
        this.renderAbyssGlow(ctx, 'left', ledgeHalfWidth, time, fogFar);
      }
    }
    
    // Render void on right side
    if (!hasRightWall) {
      const nearLedge = camera.project({ x: ledgeHalfWidth, y: 0, z: zNear });
      const farLedge = camera.project({ x: ledgeHalfWidth, y: 0, z: zFar });
      const nearVoid = camera.project({ x: ledgeHalfWidth + voidWidth, y: voidDepth, z: zNear });
      const farVoid = camera.project({ x: ledgeHalfWidth + voidWidth, y: voidDepth, z: zFar });
      
      if (nearLedge && farLedge && nearVoid && farVoid) {
        const voidGradient = ctx.ctx.createLinearGradient(
          nearLedge.x, nearLedge.y,
          nearVoid.x, this.height
        );
        
        voidGradient.addColorStop(0, color.toRgba({ ...biomeConfig.fogColor, a: 0.9 }));
        voidGradient.addColorStop(0.4, color.toRgba({ r: 3, g: 2, b: 5, a: 1 }));
        voidGradient.addColorStop(1, color.toRgba({ r: 0, g: 0, b: 0, a: 1 }));
        
        ctx.ctx.fillStyle = voidGradient;
        ctx.ctx.beginPath();
        ctx.ctx.moveTo(nearLedge.x, nearLedge.y);
        ctx.ctx.lineTo(farLedge.x, farLedge.y);
        ctx.ctx.lineTo(this.width, this.height);
        ctx.ctx.lineTo(this.width, nearVoid.y);
        ctx.ctx.closePath();
        ctx.ctx.fill();
        this.drawCalls++;
      }
      
      if (abyssGlowColor && abyssGlowIntensity > 0) {
        this.renderAbyssGlow(ctx, 'right', ledgeHalfWidth, time, fogFar);
      }
    }
    
    // Render distant rock formations for scale
    this.renderDistantFormations(ctx, chunk, hasLeftWall, hasRightWall);
  }
  
  /**
   * Render distant rock formations/pillars in the void for sense of scale
   */
  private renderDistantFormations(
    ctx: RenderContext,
    chunk: Chunk,
    hasLeftWall: boolean,
    hasRightWall: boolean
  ): void {
    const { camera, biomeConfig, time } = ctx;
    const seed = chunk.id * 31337;
    
    // Only render a few formations per chunk for performance
    const formationCount = 2;
    
    for (let i = 0; i < formationCount; i++) {
      const formationSeed = seed + i * 7919;
      const side = (formationSeed % 2 === 0) ? 'left' : 'right';
      
      // Skip if there's a wall on that side
      if ((side === 'left' && hasLeftWall) || (side === 'right' && hasRightWall)) {
        continue;
      }
      
      // Position the formation in the void
      const xOffset = 8 + (formationSeed % 100) / 10; // 8-18 units away
      const x = side === 'left' ? -xOffset : xOffset;
      const z = chunk.zStart + ((formationSeed * 3) % 100) / 100 * (chunk.zEnd - chunk.zStart);
      const baseY = -15 - (formationSeed % 30); // Deep in the void
      const height = 10 + (formationSeed % 40); // Tall formations
      
      // Project formation
      const bottom = camera.project({ x, y: baseY, z });
      const top = camera.project({ x, y: baseY + height, z });
      
      if (!bottom || !top) continue;
      
      // Calculate fog (heavier for distant formations)
      const depth = camera.getDepthFactor(z, FAR_PLANE);
      const fog = this.calculateFog(depth, biomeConfig.fogDensity * 1.5);
      
      if (fog > 0.95) continue; // Too foggy to see
      
      // Formation color - very dark, barely visible
      const baseColor = { r: 15, g: 12, b: 18 };
      const formationColor = this.applyFog(baseColor, biomeConfig.fogColor, fog);
      
      // Draw the pillar silhouette
      const pillarWidth = (3 + (formationSeed % 5)) * bottom.scale;
      
      ctx.ctx.fillStyle = color.toRgb(formationColor);
      ctx.ctx.beginPath();
      
      // Jagged pillar shape
      const widthTop = pillarWidth * 0.6;
      ctx.ctx.moveTo(bottom.x - pillarWidth / 2, bottom.y);
      ctx.ctx.lineTo(bottom.x + pillarWidth / 2, bottom.y);
      ctx.ctx.lineTo(top.x + widthTop / 2, top.y);
      ctx.ctx.lineTo(top.x - widthTop / 2, top.y);
      ctx.ctx.closePath();
      ctx.ctx.fill();
      this.drawCalls++;
      
      // Optional: Faint glow from formation (very subtle)
      if (biomeConfig.abyssGlowIntensity > 0 && (formationSeed % 5 === 0)) {
        const pulse = 0.5 + Math.sin(time * 0.3 + formationSeed) * 0.3;
        const glowAlpha = biomeConfig.abyssGlowIntensity * (1 - fog) * pulse * 0.15;
        
        if (glowAlpha > 0.01) {
          const glowGradient = ctx.ctx.createRadialGradient(
            bottom.x, bottom.y, 0,
            bottom.x, bottom.y, pillarWidth * 3
          );
          const glowColor = biomeConfig.abyssGlowColor || { r: 40, g: 20, b: 60 };
          glowGradient.addColorStop(0, color.toRgba({ ...glowColor, a: glowAlpha }));
          glowGradient.addColorStop(1, 'rgba(0, 0, 0, 0)');
          
          ctx.ctx.fillStyle = glowGradient;
          ctx.ctx.beginPath();
          ctx.ctx.arc(bottom.x, bottom.y, pillarWidth * 3, 0, Math.PI * 2);
          ctx.ctx.fill();
          this.drawCalls++;
        }
      }
    }
  }
  
  /**
   * Render subtle glow emanating from the abyss depths
   */
  private renderAbyssGlow(
    ctx: RenderContext,
    side: 'left' | 'right',
    _ledgeHalfWidth: number,
    time: number,
    fog: number
  ): void {
    const { biomeConfig } = ctx;
    const glowColor = biomeConfig.abyssGlowColor!;
    const intensity = biomeConfig.abyssGlowIntensity * (1 - fog * 0.5);
    
    // Pulsing glow effect
    const pulse = 0.7 + Math.sin(time * 0.5) * 0.2 + Math.sin(time * 0.8) * 0.1;
    const alpha = intensity * pulse * 0.3;
    
    // Position the glow
    const glowX = side === 'left' 
      ? this.width * 0.15 
      : this.width * 0.85;
    const glowY = this.height * 0.85;
    const glowRadius = this.height * 0.5;
    
    const gradient = ctx.ctx.createRadialGradient(
      glowX, glowY, 0,
      glowX, glowY, glowRadius
    );
    
    gradient.addColorStop(0, color.toRgba({ ...glowColor, a: alpha }));
    gradient.addColorStop(0.5, color.toRgba({ ...glowColor, a: alpha * 0.3 }));
    gradient.addColorStop(1, 'rgba(0, 0, 0, 0)');
    
    ctx.ctx.fillStyle = gradient;
    ctx.ctx.fillRect(
      side === 'left' ? 0 : this.width / 2,
      this.height / 2,
      this.width / 2,
      this.height / 2
    );
    this.drawCalls++;
    
    // Occasional flicker/pulse from depths (rare, ominous)
    const deepPulse = Math.sin(time * 0.3) * Math.sin(time * 0.7);
    if (deepPulse > 0.7) {
      const flickerAlpha = (deepPulse - 0.7) * intensity * 0.4;
      ctx.ctx.fillStyle = color.toRgba({ ...glowColor, a: flickerAlpha });
      ctx.ctx.fillRect(
        side === 'left' ? 0 : this.width * 0.6,
        this.height * 0.7,
        this.width * 0.4,
        this.height * 0.3
      );
      this.drawCalls++;
    }
  }
  
  /**
   * Render the visible edge/side of the ledge
   */
  private renderLedgeEdge(
    ctx: RenderContext,
    zNear: number,
    zFar: number,
    ledgeHalfWidth: number,
    hasLeftWall: boolean,
    hasRightWall: boolean,
    fogNear: number,
    fogFar: number,
    flicker: number
  ): void {
    const { camera, biomeConfig } = ctx;
    const ledgeDepth = 0.4; // Visual thickness of the ledge
    
    const edgeColor = { r: 25, g: 20, b: 18 }; // Darker rock color for edge
    
    // Left edge (visible when there's a void on the left)
    if (!hasLeftWall) {
      const topNear = camera.project({ x: -ledgeHalfWidth, y: 0, z: zNear });
      const topFar = camera.project({ x: -ledgeHalfWidth, y: 0, z: zFar });
      const bottomNear = camera.project({ x: -ledgeHalfWidth, y: -ledgeDepth, z: zNear });
      const bottomFar = camera.project({ x: -ledgeHalfWidth, y: -ledgeDepth, z: zFar });
      
      if (topNear && topFar && bottomNear && bottomFar) {
        const colorNear = this.applyFogAndLight(edgeColor, biomeConfig.fogColor, fogNear, flicker * 0.7);
        const colorFar = this.applyFogAndLight(edgeColor, biomeConfig.fogColor, fogFar, flicker * 0.7);
        
        this.drawQuadGradient(ctx.ctx, topFar, bottomFar, bottomNear, topNear, colorFar, colorNear);
      }
    }
    
    // Right edge
    if (!hasRightWall) {
      const topNear = camera.project({ x: ledgeHalfWidth, y: 0, z: zNear });
      const topFar = camera.project({ x: ledgeHalfWidth, y: 0, z: zFar });
      const bottomNear = camera.project({ x: ledgeHalfWidth, y: -ledgeDepth, z: zNear });
      const bottomFar = camera.project({ x: ledgeHalfWidth, y: -ledgeDepth, z: zFar });
      
      if (topNear && topFar && bottomNear && bottomFar) {
        const colorNear = this.applyFogAndLight(edgeColor, biomeConfig.fogColor, fogNear, flicker * 0.7);
        const colorFar = this.applyFogAndLight(edgeColor, biomeConfig.fogColor, fogFar, flicker * 0.7);
        
        this.drawQuadGradient(ctx.ctx, topFar, bottomFar, bottomNear, topNear, colorFar, colorNear);
      }
    }
  }
  
  /**
   * Render subtle rock texture details on walls
   */
  private renderWallDetails(
    ctx: RenderContext,
    zNear: number,
    zFar: number,
    halfWidth: number,
    tunnelHeight: number,
    fog: number
  ): void {
    const { camera, biomeConfig } = ctx;
    const detailAlpha = (1 - fog) * 0.15;
    
    // Use chunk-based seed for consistent details
    const seed = Math.floor(zNear / 10) * 1337;
    
    // Rock cracks/details on left wall (only if wall exists)
    if (biomeConfig.hasLeftWall) {
      for (let i = 0; i < 3; i++) {
        const crackSeed = seed + i * 7;
        const crackZ = zNear + ((crackSeed % 100) / 100) * (zFar - zNear);
        const crackY = ((crackSeed * 3) % 100) / 100 * tunnelHeight * 0.7 + 0.3;
        
        const start = camera.project({ x: -halfWidth + 0.02, y: crackY, z: crackZ });
        const end = camera.project({ 
          x: -halfWidth + 0.02, 
          y: crackY + ((crackSeed % 50) / 100 - 0.25) * 0.8, 
          z: crackZ + ((crackSeed * 2) % 30) / 30 * 2 - 1 
        });
        
        if (start && end) {
          ctx.ctx.strokeStyle = `rgba(20, 15, 12, ${detailAlpha})`;
          ctx.ctx.lineWidth = 1;
          ctx.ctx.beginPath();
          ctx.ctx.moveTo(start.x, start.y);
          ctx.ctx.lineTo(end.x, end.y);
          ctx.ctx.stroke();
          this.drawCalls++;
        }
      }
    }
    
    // Rock cracks on right wall (only if wall exists)
    if (biomeConfig.hasRightWall) {
      for (let i = 0; i < 3; i++) {
        const crackSeed = seed + i * 11 + 500;
        const crackZ = zNear + ((crackSeed % 100) / 100) * (zFar - zNear);
        const crackY = ((crackSeed * 3) % 100) / 100 * tunnelHeight * 0.7 + 0.3;
        
        const start = camera.project({ x: halfWidth - 0.02, y: crackY, z: crackZ });
        const end = camera.project({ 
          x: halfWidth - 0.02, 
          y: crackY + ((crackSeed % 50) / 100 - 0.25) * 0.8, 
          z: crackZ + ((crackSeed * 2) % 30) / 30 * 2 - 1 
        });
        
        if (start && end) {
          ctx.ctx.strokeStyle = `rgba(20, 15, 12, ${detailAlpha})`;
          ctx.ctx.lineWidth = 1;
          ctx.ctx.beginPath();
          ctx.ctx.moveTo(start.x, start.y);
          ctx.ctx.lineTo(end.x, end.y);
          ctx.ctx.stroke();
          this.drawCalls++;
        }
      }
    }
  }
  
  /**
   * Render the track (rails and ties)
   */
  private renderTrack(ctx: RenderContext, chunk: Chunk): void {
    const { camera, biomeConfig } = ctx;
    const railY = 0.05;
    const railWidth = 0.08;
    const halfSpacing = RAIL_SPACING / 2;
    
    const segments = 8;
    const segmentLength = (chunk.zEnd - chunk.zStart) / segments;
    
    for (let i = segments - 1; i >= 0; i--) {
      const zFar = chunk.zStart + (i + 1) * segmentLength;
      const zNear = chunk.zStart + i * segmentLength;
      
      const depth = camera.getDepthFactor((zFar + zNear) / 2, FAR_PLANE);
      const fog = this.calculateFog(depth, biomeConfig.fogDensity);
      const railColor = this.applyFog(biomeConfig.railColor, biomeConfig.fogColor, fog);
      
      // Left rail
      const leftNear1 = camera.project({ x: -halfSpacing - railWidth, y: railY, z: zNear });
      const leftNear2 = camera.project({ x: -halfSpacing + railWidth, y: railY, z: zNear });
      const leftFar1 = camera.project({ x: -halfSpacing - railWidth, y: railY, z: zFar });
      const leftFar2 = camera.project({ x: -halfSpacing + railWidth, y: railY, z: zFar });
      
      if (leftNear1 && leftNear2 && leftFar1 && leftFar2) {
        this.drawQuad(ctx.ctx, leftFar1, leftFar2, leftNear2, leftNear1, railColor);
      }
      
      // Right rail
      const rightNear1 = camera.project({ x: halfSpacing - railWidth, y: railY, z: zNear });
      const rightNear2 = camera.project({ x: halfSpacing + railWidth, y: railY, z: zNear });
      const rightFar1 = camera.project({ x: halfSpacing - railWidth, y: railY, z: zFar });
      const rightFar2 = camera.project({ x: halfSpacing + railWidth, y: railY, z: zFar });
      
      if (rightNear1 && rightNear2 && rightFar1 && rightFar2) {
        this.drawQuad(ctx.ctx, rightFar1, rightFar2, rightNear2, rightNear1, railColor);
      }
    }
  }
  
  /**
   * Render a mine object
   */
  private renderObject(ctx: RenderContext, obj: MineObject): void {
    const { camera, biomeConfig } = ctx;
    const projected = camera.project(obj.position);
    
    if (!projected) return;
    
    const depth = camera.getDepthFactor(obj.position.z, FAR_PLANE);
    const fog = this.calculateFog(depth, biomeConfig.fogDensity);
    
    // Don't render objects that are too foggy
    if (fog > 0.95) return;
    
    this.objectsRendered++;
    
    switch (obj.type) {
      case 'support_beam':
        this.renderSupportBeam(ctx, obj, projected, fog);
        break;
      case 'lantern':
        this.renderLantern(ctx, obj, projected, fog);
        break;
      case 'rail_tie':
        this.renderRailTie(ctx, obj, projected, fog);
        break;
      case 'stalactite':
        this.renderStalactite(ctx, obj, projected, fog);
        break;
      case 'rubble':
        this.renderRubble(ctx, obj, projected, fog);
        break;
      case 'chain':
        this.renderChain(ctx, obj, projected, fog);
        break;
      default:
        // Generic fallback
        break;
    }
  }
  
  private renderSupportBeam(
    ctx: RenderContext,
    _obj: MineObject,
    projected: { x: number; y: number; scale: number },
    fog: number
  ): void {
    const { biomeConfig } = ctx;
    const beamColor = this.applyFog({ r: 60, g: 45, b: 30 }, biomeConfig.fogColor, fog);
    
    const beamWidth = 15 * projected.scale;
    const beamHeight = biomeConfig.tunnelHeight * projected.scale * 50;
    
    ctx.ctx.fillStyle = color.toRgb(beamColor);
    ctx.ctx.fillRect(
      projected.x - beamWidth / 2,
      projected.y - beamHeight,
      beamWidth,
      beamHeight
    );
    this.drawCalls++;
  }
  
  private renderLantern(
    ctx: RenderContext,
    obj: MineObject,
    projected: { x: number; y: number; scale: number },
    fog: number
  ): void {
    const { biomeConfig, time } = ctx;
    
    // Multiple flicker frequencies for realistic fire
    const flicker1 = Math.sin(time * 8 + obj.id) * 0.15;
    const flicker2 = Math.sin(time * 13 + obj.id * 1.3) * 0.1;
    const flicker3 = Math.sin(time * 21 + obj.id * 0.7) * 0.05;
    const flicker = 0.75 + flicker1 + flicker2 + flicker3;
    
    const size = 10 * projected.scale * obj.scale;
    
    // Outer glow (large, soft)
    const outerGlowSize = size * 6 * flicker;
    const outerGlowAlpha = (1 - fog) * 0.25 * flicker;
    
    const outerGradient = ctx.ctx.createRadialGradient(
      projected.x, projected.y, 0,
      projected.x, projected.y, outerGlowSize
    );
    outerGradient.addColorStop(0, `rgba(255, 150, 50, ${outerGlowAlpha})`);
    outerGradient.addColorStop(0.4, `rgba(255, 100, 30, ${outerGlowAlpha * 0.4})`);
    outerGradient.addColorStop(1, 'rgba(200, 60, 10, 0)');
    
    ctx.ctx.fillStyle = outerGradient;
    ctx.ctx.beginPath();
    ctx.ctx.arc(projected.x, projected.y, outerGlowSize, 0, Math.PI * 2);
    ctx.ctx.fill();
    
    // Inner glow (bright core)
    const innerGlowSize = size * 2.5 * flicker;
    const innerGlowAlpha = (1 - fog) * 0.7 * flicker;
    
    const innerGradient = ctx.ctx.createRadialGradient(
      projected.x, projected.y, 0,
      projected.x, projected.y, innerGlowSize
    );
    innerGradient.addColorStop(0, `rgba(255, 240, 200, ${innerGlowAlpha})`);
    innerGradient.addColorStop(0.3, `rgba(255, 180, 80, ${innerGlowAlpha * 0.7})`);
    innerGradient.addColorStop(1, 'rgba(255, 100, 30, 0)');
    
    ctx.ctx.fillStyle = innerGradient;
    ctx.ctx.beginPath();
    ctx.ctx.arc(projected.x, projected.y, innerGlowSize, 0, Math.PI * 2);
    ctx.ctx.fill();
    
    // Lantern body (dark metal frame)
    const lanternColor = this.applyFog({ r: 35, g: 30, b: 25 }, biomeConfig.fogColor, fog);
    ctx.ctx.fillStyle = color.toRgb(lanternColor);
    ctx.ctx.fillRect(projected.x - size / 2, projected.y - size * 0.7, size, size * 1.4);
    
    // Bright center (flame)
    const flameSize = size * 0.4;
    ctx.ctx.fillStyle = `rgba(255, 220, 150, ${(1 - fog) * 0.9})`;
    ctx.ctx.beginPath();
    ctx.ctx.arc(projected.x, projected.y, flameSize, 0, Math.PI * 2);
    ctx.ctx.fill();
    
    this.drawCalls += 4;
  }
  
  private renderRailTie(
    ctx: RenderContext,
    obj: MineObject,
    projected: { x: number; y: number; scale: number },
    fog: number
  ): void {
    const { biomeConfig, camera } = ctx;
    const tieColor = this.applyFog(biomeConfig.trackColor, biomeConfig.fogColor, fog);
    
    // Project the tie ends
    const tieLeft = camera.project({ x: -RAIL_SPACING / 2 - 0.3, y: 0.02, z: obj.position.z });
    const tieRight = camera.project({ x: RAIL_SPACING / 2 + 0.3, y: 0.02, z: obj.position.z });
    
    if (!tieLeft || !tieRight) return;
    
    const tieHeight = 4 * projected.scale;
    
    ctx.ctx.fillStyle = color.toRgb(tieColor);
    ctx.ctx.beginPath();
    ctx.ctx.moveTo(tieLeft.x, tieLeft.y);
    ctx.ctx.lineTo(tieRight.x, tieRight.y);
    ctx.ctx.lineTo(tieRight.x, tieRight.y + tieHeight);
    ctx.ctx.lineTo(tieLeft.x, tieLeft.y + tieHeight);
    ctx.ctx.closePath();
    ctx.ctx.fill();
    this.drawCalls++;
  }
  
  private renderStalactite(
    ctx: RenderContext,
    obj: MineObject,
    projected: { x: number; y: number; scale: number },
    fog: number
  ): void {
    const { biomeConfig } = ctx;
    const stoneColor = this.applyFog({ r: 50, g: 45, b: 40 }, biomeConfig.fogColor, fog);
    
    const width = 8 * projected.scale * obj.scale;
    const height = 25 * projected.scale * obj.scale;
    
    ctx.ctx.fillStyle = color.toRgb(stoneColor);
    ctx.ctx.beginPath();
    ctx.ctx.moveTo(projected.x - width / 2, projected.y);
    ctx.ctx.lineTo(projected.x + width / 2, projected.y);
    ctx.ctx.lineTo(projected.x, projected.y + height);
    ctx.ctx.closePath();
    ctx.ctx.fill();
    this.drawCalls++;
  }
  
  private renderRubble(
    ctx: RenderContext,
    obj: MineObject,
    projected: { x: number; y: number; scale: number },
    fog: number
  ): void {
    const { biomeConfig } = ctx;
    const rubbleColor = this.applyFog({ r: 45, g: 40, b: 35 }, biomeConfig.fogColor, fog);
    
    const size = 10 * projected.scale * obj.scale;
    
    ctx.ctx.fillStyle = color.toRgb(rubbleColor);
    ctx.ctx.beginPath();
    ctx.ctx.arc(projected.x, projected.y, size, 0, Math.PI * 2);
    ctx.ctx.fill();
    this.drawCalls++;
  }
  
  private renderChain(
    ctx: RenderContext,
    obj: MineObject,
    projected: { x: number; y: number; scale: number },
    fog: number
  ): void {
    const { camera } = ctx;
    const chainColor = this.applyFog({ r: 70, g: 65, b: 60 }, ctx.biomeConfig.fogColor, fog);
    
    const chainLength = 1.5 * obj.scale;
    const chainBottom = camera.project({ 
      x: obj.position.x, 
      y: obj.position.y - chainLength, 
      z: obj.position.z 
    });
    
    if (!chainBottom) return;
    
    ctx.ctx.strokeStyle = color.toRgb(chainColor);
    ctx.ctx.lineWidth = 2 * projected.scale;
    ctx.ctx.beginPath();
    ctx.ctx.moveTo(projected.x, projected.y);
    ctx.ctx.lineTo(chainBottom.x, chainBottom.y);
    ctx.ctx.stroke();
    this.drawCalls++;
  }
  
  /**
   * Render screen-space atmospheric effects
   */
  private renderAtmosphericEffects(ctx: RenderContext): void {
    const { time } = ctx;
    
    // Vignette effect - darker and more dramatic
    const gradient = ctx.ctx.createRadialGradient(
      this.width / 2, this.height / 2, this.height * 0.2,
      this.width / 2, this.height / 2, this.height * 0.85
    );
    gradient.addColorStop(0, 'rgba(0, 0, 0, 0)');
    gradient.addColorStop(0.6, 'rgba(0, 0, 0, 0.3)');
    gradient.addColorStop(1, 'rgba(0, 0, 0, 0.7)');
    
    ctx.ctx.fillStyle = gradient;
    ctx.ctx.fillRect(0, 0, this.width, this.height);
    this.drawCalls++;
    
    // Subtle dust particles floating
    const particleCount = 20;
    
    for (let i = 0; i < particleCount; i++) {
      const phase = (time * 0.2 + i * 0.37) % 1;
      const xWave = Math.sin(time * 0.3 + i * 2.1) * 0.15;
      const x = ((i / particleCount) + xWave + 0.5) % 1 * this.width;
      const y = phase * this.height;
      const size = 0.8 + Math.sin(i * 1.7) * 0.4;
      const alpha = 0.08 + Math.sin(time * 2 + i) * 0.04;
      
      ctx.ctx.fillStyle = `rgba(180, 160, 130, ${alpha})`;
      ctx.ctx.beginPath();
      ctx.ctx.arc(x, y, size, 0, Math.PI * 2);
      ctx.ctx.fill();
    }
    this.drawCalls++;
    
    // Occasional flicker overlay
    const flickerIntensity = Math.sin(time * 7) * Math.sin(time * 11) * 0.015;
    if (flickerIntensity > 0) {
      ctx.ctx.fillStyle = `rgba(255, 200, 100, ${flickerIntensity})`;
      ctx.ctx.fillRect(0, 0, this.width, this.height);
      this.drawCalls++;
    }
    
    // Top and bottom edge darkening
    const edgeGradientTop = ctx.ctx.createLinearGradient(0, 0, 0, this.height * 0.15);
    edgeGradientTop.addColorStop(0, 'rgba(0, 0, 0, 0.4)');
    edgeGradientTop.addColorStop(1, 'rgba(0, 0, 0, 0)');
    ctx.ctx.fillStyle = edgeGradientTop;
    ctx.ctx.fillRect(0, 0, this.width, this.height * 0.15);
    
    const edgeGradientBottom = ctx.ctx.createLinearGradient(0, this.height * 0.85, 0, this.height);
    edgeGradientBottom.addColorStop(0, 'rgba(0, 0, 0, 0)');
    edgeGradientBottom.addColorStop(1, 'rgba(0, 0, 0, 0.3)');
    ctx.ctx.fillStyle = edgeGradientBottom;
    ctx.ctx.fillRect(0, this.height * 0.85, this.width, this.height * 0.15);
    this.drawCalls += 2;
  }
  
  // ===== Helper Methods =====
  
  private calculateFog(depth: number, density: number): number {
    // Exponential fog
    return 1 - Math.exp(-density * depth * 30);
  }
  
  private applyFog(baseColor: Color, fogColor: Color, fogAmount: number): Color {
    return color.lerp(baseColor, fogColor, clamp(fogAmount, 0, 1));
  }
  
  private applyFogAndLight(baseColor: Color, fogColor: Color, fogAmount: number, light: number): Color {
    const lit = color.darken(baseColor, 1 - light);
    return this.applyFog(lit, fogColor, fogAmount);
  }
  
  private drawQuad(
    ctx: CanvasRenderingContext2D,
    p1: { x: number; y: number },
    p2: { x: number; y: number },
    p3: { x: number; y: number },
    p4: { x: number; y: number },
    fillColor: Color
  ): void {
    ctx.fillStyle = color.toRgb(fillColor);
    ctx.beginPath();
    ctx.moveTo(p1.x, p1.y);
    ctx.lineTo(p2.x, p2.y);
    ctx.lineTo(p3.x, p3.y);
    ctx.lineTo(p4.x, p4.y);
    ctx.closePath();
    ctx.fill();
    this.drawCalls++;
  }
  
  private drawQuadGradient(
    ctx: CanvasRenderingContext2D,
    farLeft: { x: number; y: number },
    farRight: { x: number; y: number },
    nearRight: { x: number; y: number },
    nearLeft: { x: number; y: number },
    farColor: Color,
    nearColor: Color
  ): void {
    // For performance, just use the average color
    const avgColor = color.lerp(farColor, nearColor, 0.5);
    this.drawQuad(ctx, farLeft, farRight, nearRight, nearLeft, avgColor);
  }
  
  getStats(): { drawCalls: number; objectsRendered: number } {
    return {
      drawCalls: this.drawCalls,
      objectsRendered: this.objectsRendered,
    };
  }
  
  getSize(): { width: number; height: number } {
    return { width: this.width, height: this.height };
  }
