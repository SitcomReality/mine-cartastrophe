/**
 * Lightweight deterministic noise utilities.
 * No external dependencies. Designed for speed on low-end hardware.
 */

/** Fast seedable pseudo-random (mulberry32) */
export function mulberry32(seed: number) {
  return function (): number {
    seed |= 0;
    seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 0xffffffff;
  };
}

/**
 * 1-D smooth noise via cubic-interpolated lookup.
 * Returns a value in [-1, 1].
 */
export function smoothNoise1D(x: number, seed: number): number {
  const ix = Math.floor(x);
  const fx = x - ix;
  // smooth step
  const u = fx * fx * (3 - 2 * fx);

  const rand = (n: number) => {
    // fast hash
    let h = (n ^ seed) >>> 0;
    h = Math.imul(h ^ (h >>> 16), 0x45d9f3b);
    h = Math.imul(h ^ (h >>> 16), 0x45d9f3b);
    h = h ^ (h >>> 16);
    return (h >>> 0) / 0xffffffff * 2 - 1;
  };

  return rand(ix) * (1 - u) + rand(ix + 1) * u;
}

/**
 * Fractional Brownian Motion — layered octaves of smoothNoise1D.
 * Returns value in roughly [-1, 1].
 */
export function fbm1D(
  x: number,
  seed: number,
  octaves = 4,
  lacunarity = 2.0,
  gain = 0.5,
): number {
  let value = 0;
  let amplitude = 1;
  let frequency = 1;
  let maxValue = 0;

  for (let i = 0; i < octaves; i++) {
    value += smoothNoise1D(x * frequency, seed + i * 1000) * amplitude;
    maxValue += amplitude;
    amplitude *= gain;
    frequency *= lacunarity;
  }

  return value / maxValue;
}
