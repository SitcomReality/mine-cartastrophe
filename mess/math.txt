import type { Vec3, Vec2, Color } from '@/types';

// Vector operations
export const vec3 = {
  create: (x = 0, y = 0, z = 0): Vec3 => ({ x, y, z }),
  
  add: (a: Vec3, b: Vec3): Vec3 => ({
    x: a.x + b.x,
    y: a.y + b.y,
    z: a.z + b.z,
  }),
  
  sub: (a: Vec3, b: Vec3): Vec3 => ({
    x: a.x - b.x,
    y: a.y - b.y,
    z: a.z - b.z,
  }),
  
  scale: (v: Vec3, s: number): Vec3 => ({
    x: v.x * s,
    y: v.y * s,
    z: v.z * s,
  }),
  
  length: (v: Vec3): number => 
    Math.sqrt(v.x * v.x + v.y * v.y + v.z * v.z),
  
  normalize: (v: Vec3): Vec3 => {
    const len = vec3.length(v);
    if (len === 0) return { x: 0, y: 0, z: 0 };
    return vec3.scale(v, 1 / len);
  },
  
  lerp: (a: Vec3, b: Vec3, t: number): Vec3 => ({
    x: a.x + (b.x - a.x) * t,
    y: a.y + (b.y - a.y) * t,
    z: a.z + (b.z - a.z) * t,
  }),
  
  clone: (v: Vec3): Vec3 => ({ x: v.x, y: v.y, z: v.z }),
};

export const vec2 = {
  create: (x = 0, y = 0): Vec2 => ({ x, y }),
};

// Color utilities
export const color = {
  toRgba: (c: Color, alpha?: number): string => {
    const a = alpha ?? c.a ?? 1;
    return `rgba(${Math.round(c.r)}, ${Math.round(c.g)}, ${Math.round(c.b)}, ${a})`;
  },
  
  toRgb: (c: Color): string => 
    `rgb(${Math.round(c.r)}, ${Math.round(c.g)}, ${Math.round(c.b)})`,
  
  lerp: (a: Color, b: Color, t: number): Color => ({
    r: a.r + (b.r - a.r) * t,
    g: a.g + (b.g - a.g) * t,
    b: a.b + (b.b - a.b) * t,
    a: (a.a ?? 1) + ((b.a ?? 1) - (a.a ?? 1)) * t,
  }),
  
  darken: (c: Color, amount: number): Color => ({
    r: c.r * (1 - amount),
    g: c.g * (1 - amount),
    b: c.b * (1 - amount),
    a: c.a,
  }),
  
  brighten: (c: Color, amount: number): Color => ({
    r: Math.min(255, c.r + (255 - c.r) * amount),
    g: Math.min(255, c.g + (255 - c.g) * amount),
    b: Math.min(255, c.b + (255 - c.b) * amount),
    a: c.a,
  }),
};

// Math utilities
export const clamp = (value: number, min: number, max: number): number =>
  Math.max(min, Math.min(max, value));

export const lerp = (a: number, b: number, t: number): number =>
  a + (b - a) * t;

export const smoothstep = (edge0: number, edge1: number, x: number): number => {
  const t = clamp((x - edge0) / (edge1 - edge0), 0, 1);
  return t * t * (3 - 2 * t);
};

export const degToRad = (degrees: number): number =>
  degrees * (Math.PI / 180);

export const radToDeg = (radians: number): number =>
  radians * (180 / Math.PI);

// Simple seeded random number generator (Mulberry32)
export const createRNG = (seed: number) => {
  let state = seed;
  
  return {
    next: (): number => {
      state |= 0;
      state = (state + 0x6d2b79f5) | 0;
      let t = Math.imul(state ^ (state >>> 15), 1 | state);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    },
    
    range: (min: number, max: number): number => {
      state |= 0;
      state = (state + 0x6d2b79f5) | 0;
      let t = Math.imul(state ^ (state >>> 15), 1 | state);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      const rand = ((t ^ (t >>> 14)) >>> 0) / 4294967296;
      return min + rand * (max - min);
    },
    
    int: (min: number, max: number): number => {
      state |= 0;
      state = (state + 0x6d2b79f5) | 0;
      let t = Math.imul(state ^ (state >>> 15), 1 | state);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      const rand = ((t ^ (t >>> 14)) >>> 0) / 4294967296;
      return Math.floor(min + rand * (max - min + 1));
    },
    
    chance: (probability: number): boolean => {
      state |= 0;
      state = (state + 0x6d2b79f5) | 0;
      let t = Math.imul(state ^ (state >>> 15), 1 | state);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      const rand = ((t ^ (t >>> 14)) >>> 0) / 4294967296;
      return rand < probability;
    },
    
    setSeed: (newSeed: number) => {
      state = newSeed;
    },
  };
};

// Simple 1D noise function
export const noise1D = (x: number, seed: number = 0): number => {
  const rng = createRNG(Math.floor(x) + seed * 10000);
  const rng2 = createRNG(Math.floor(x) + 1 + seed * 10000);
  const a = rng.next();
  const b = rng2.next();
  const t = x - Math.floor(x);
  return lerp(a, b, smoothstep(0, 1, t));
};
