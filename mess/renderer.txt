/**
 * Mine Cartastrophe — Canvas 2D Renderer
 *
 * Technique: column-by-column raycasting / pseudo-3D tunnel projection.
 *
 * For each vertical column on screen we cast a ray forward into the tunnel
 * and compute where it hits the left wall, right wall, ceiling, and floor.
 * The result is a set of vertical spans that we paint with shaded colors.
 *
 * Performance strategy:
 *  - Render at a reduced internal resolution (configurable scale factor)
 *  - Use integer pixel ops, avoid per-pixel JS loops where possible
 *  - Reuse ImageData buffer between frames
 *  - All math is pre-computed per column, not per pixel
 */

import type { WorldState } from './world';
import type { BiomeConfig } from '../biomes/types';
import { sampleTrack } from './track';
import { fbm1D, smoothNoise1D } from './noise';

// ─── Render Resolution ───────────────────────────────────────────────────────
// The canvas is rendered at SCALE * display pixels then CSS-scaled up.
// 0.5 = half-res (huge perf win, chunky retro look).
export const RENDER_SCALE = 0.5;

// ─── Projection Constants ─────────────────────────────────────────────────────
const FOV_FACTOR = 0.75;          // horizontal FOV multiplier (lower = wider)
const MAX_DEPTH = 32;              // world units before fog fully takes over
const RAIL_GAUGE = 0.28;           // half-distance between rails (NDC)
const TIE_SPACING = 1.2;          // world units between sleepers
const FLOOR_Y_NDC = 0.05;         // NDC of floor-center (slightly below mid)
const CAMERA_HEIGHT = 0.55;       // 0=floor 1=ceiling, eye position

// ─── Colour helpers ───────────────────────────────────────────────────────────
function hexToRgb(hex: string): [number, number, number] {
  const h = hex.replace('#', '');
  const n = parseInt(h, 16);
  return [(n >> 16) & 0xff, (n >> 8) & 0xff, n & 0xff];
}

function lerpColor(
  a: [number, number, number],
  b: [number, number, number],
  t: number,
): [number, number, number] {
  t = Math.max(0, Math.min(1, t));
  return [
    (a[0] + (b[0] - a[0]) * t) | 0,
    (a[1] + (b[1] - a[1]) * t) | 0,
    (a[2] + (b[2] - a[2]) * t) | 0,
  ];
}

function addLight(
  base: [number, number, number],
  glow: [number, number, number],
  intensity: number,
): [number, number, number] {
  return [
    Math.min(255, (base[0] + glow[0] * intensity) | 0),
    Math.min(255, (base[1] + glow[1] * intensity) | 0),
    Math.min(255, (base[2] + glow[2] * intensity) | 0),
  ];
}

// ─── Torch ───────────────────────────────────────────────────────────────────
interface TorchInfo {
  distAlongTrack: number; // world-unit distance of this torch
  side: number;           // -1 = left, +1 = right
  flicker: number;        // cached flicker value (updated per frame)
}

const MAX_TORCHES = 12; // torches kept in a circular buffer

// ─── Main renderer class ──────────────────────────────────────────────────────
export class MineRenderer {
  private canvas: HTMLCanvasElement;
  private ctx: CanvasRenderingContext2D;
  private offscreen: OffscreenCanvas | null = null;
  private offCtx: OffscreenCanvasRenderingContext2D | null = null;
  private buf: ImageData | null = null;
  private rw = 0; // render width
  private rh = 0; // render height

  private torches: TorchInfo[] = [];

  // Particle systems
  private dustParticles: Float32Array = new Float32Array(0); // x,y,z,speed per particle
  private dustCount = 0;

  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d', { alpha: false })!;
    this.ctx.imageSmoothingEnabled = false;
  }

  // Call whenever the canvas display size changes
  resize(displayW: number, displayH: number) {
    this.rw = Math.ceil(displayW * RENDER_SCALE) | 0;
    this.rh = Math.ceil(displayH * RENDER_SCALE) | 0;

    this.canvas.width = displayW;
    this.canvas.height = displayH;

    if (typeof OffscreenCanvas !== 'undefined') {
      this.offscreen = new OffscreenCanvas(this.rw, this.rh);
      this.offCtx = this.offscreen.getContext('2d', { alpha: false }) as OffscreenCanvasRenderingContext2D;
      if (this.offCtx) this.offCtx.imageSmoothingEnabled = false;
      this.buf = this.offCtx!.createImageData(this.rw, this.rh);
    } else {
      this.buf = new ImageData(this.rw, this.rh);
    }

    this.initDust();
  }

  private initDust() {
    this.dustCount = 80;
    this.dustParticles = new Float32Array(this.dustCount * 4);
    for (let i = 0; i < this.dustCount; i++) {
      this.resetDust(i, Math.random() * MAX_DEPTH);
    }
  }

  private resetDust(i: number, z: number) {
    const b = i * 4;
    this.dustParticles[b]     = (Math.random() - 0.5) * 2;   // x NDC
    this.dustParticles[b + 1] = (Math.random() - 0.5) * 2;   // y NDC
    this.dustParticles[b + 2] = z;                             // depth (0=near)
    this.dustParticles[b + 3] = 0.3 + Math.random() * 0.4;    // drift speed
  }

  // ─── Main render entry point ────────────────────────────────────────────────
  render(state: WorldState) {
    if (!this.buf || this.rw === 0) return;

    const { biome, distance, elapsed, speed } = state;

    // ── Track sample at current position ───────────────────────────────────
    const track = sampleTrack(distance, biome);

    // Subtle camera bob linked to speed
    const bobY = Math.sin(elapsed * 4.5 * (speed / 10)) * 0.012;
    const bobX = Math.cos(elapsed * 2.2 * (speed / 10)) * 0.006;

    // Camera lean into curves
    const leanX = -track.dX * 0.18;

    // Effective camera offset (NDC)
    const camX = track.xOffset * 0.12 + leanX + bobX;
    const camY = track.yOffset * 0.08 + bobY;

    // ── Precompute colours ──────────────────────────────────────────────────
    const c = biome.colors;
    const fogRgb       = hexToRgb(c.fog);
    const wallNearRgb  = hexToRgb(c.wallNear);
    const wallFarRgb   = hexToRgb(c.wallFar);
    const ceilRgb      = hexToRgb(c.ceiling);
    const floorRgb     = hexToRgb(c.floor);
    const glowPrimRgb  = hexToRgb(c.glowPrimary);
    const railRgb      = hexToRgb(c.trackRail);
    const tieRgb       = hexToRgb(c.trackTie);

    // ── Update torches ──────────────────────────────────────────────────────
    this.updateTorches(distance, biome, elapsed);

    // ── Fill pixel buffer ────────────────────────────────────────────────────
    const data = this.buf.data;
    const W = this.rw;
    const H = this.rh;
    const halfH = H * 0.5;

    // Horizon line in pixels (shifted by vertical track offset & bob)
    const horizonY = (halfH * (1 + (camY + FLOOR_Y_NDC) * 1.5)) | 0;

    for (let px = 0; px < W; px++) {
      // Ray direction in NDC (-1..1)
      const rayNDC = (px / W) * 2 - 1; // -1=left, +1=right

      // Apply camera offset
      const rayX = (rayNDC - camX) / FOV_FACTOR;

      // ── Tunnel cross-section at various depths ──────────────────────────
      // We compute the column as a series of horizontal bands:
      //   [top of canvas] sky/void → ceiling → [tunnelTop] → wall → [tunnelBot] → floor → [bottom]
      //
      // The tunnel "mouth" in screen space shrinks with depth.
      // We use simple perspective: screenY = horizonY + worldY / depth * scaleY

      // Find the depth where this ray hits the side wall
      // Tunnel half-width at depth d = biome.tunnelWidth (world units)
      // Ray hits wall when |rayX| * d = tunnelWidth → d = tunnelWidth / |rayX|
      const absRayX = Math.abs(rayX);
      const wallDepth = absRayX < 0.001 ? MAX_DEPTH : Math.min(biome.tunnelWidth / absRayX, MAX_DEPTH);

      // Fog factor at wall depth
      const wallFog = fogFactor(wallDepth, biome.fogDensity);

      // Wall roughness: perturb wallDepth with noise
      const roughOff = fbm1D(
        px * 0.07 + distance * 0.03,
        biome.seedOffset + 77,
        2,
      ) * biome.tunnelRoughness * 0.5;

      // Where does the wall appear on screen?
      const perspScale = H / (wallDepth + 0.001);

      // Tunnel half-height in screen pixels at wall depth
      const tunnelPxHalf = biome.tunnelHeight * perspScale * (1 + roughOff * 0.4);

      // Y positions (screen pixels)
      const wallTop    = Math.max(0, horizonY - tunnelPxHalf) | 0;
      const wallBottom = Math.min(H - 1, horizonY + tunnelPxHalf) | 0;

      // Wall colour
      const wallBaseColor = lerpColor(wallNearRgb, wallFarRgb, wallFog);

      // Torch influence on this column
      let torchIntensity = this.computeTorchLight(distance, wallDepth, px / W, elapsed, biome);

      const wallColor = addLight(
        lerpColor(wallBaseColor, fogRgb, wallFog),
        glowPrimRgb,
        torchIntensity * 0.6,
      );

      // ── Ceiling ────────────────────────────────────────────────────────
      for (let py = 0; py < wallTop; py++) {
        // Depth of this ceiling pixel via reverse perspective
        const ndcY = (horizonY - py) / (H * 0.5);
        const depth = Math.min(biome.tunnelHeight / Math.max(0.001, ndcY) * 0.5, MAX_DEPTH);
        const fog   = fogFactor(depth, biome.fogDensity);
        const col   = lerpColor(lerpColor(ceilRgb, fogRgb, fog), fogRgb, fog * 0.5);
        const tLight = this.computeTorchLight(distance, depth, px / W, elapsed, biome);
        const lit   = addLight(col, glowPrimRgb, tLight * 0.35);
        setPixel(data, px, py, W, lit, 255);
      }

      // ── Floor ──────────────────────────────────────────────────────────
      for (let py = wallBottom + 1; py < H; py++) {
        const ndcY = (py - horizonY) / (H * 0.5);
        const depth = Math.min(biome.tunnelHeight / Math.max(0.001, ndcY) * 0.5, MAX_DEPTH);
        const fog   = fogFactor(depth, biome.fogDensity);

        // Rail & tie check
        const floorU = (rayNDC - camX);  // lateral world pos at floor
        const isRail = Math.abs(Math.abs(floorU) - RAIL_GAUGE) < 0.022 / (depth + 0.5);
        const tieZ   = distance + depth;
        const isTie  = (tieZ % TIE_SPACING < 0.18) && (Math.abs(floorU) < RAIL_GAUGE + 0.04);

        let col: [number, number, number];
        if (isRail) {
          col = lerpColor(railRgb, fogRgb, fog);
        } else if (isTie) {
          col = lerpColor(tieRgb, fogRgb, fog);
        } else {
          col = lerpColor(floorRgb, fogRgb, fog);
        }

        const tLight = this.computeTorchLight(distance, depth, px / W, elapsed, biome);
        const lit    = addLight(col, glowPrimRgb, tLight * 0.4);
        setPixel(data, px, py, W, lit, 255);
      }

      // ── Wall span ──────────────────────────────────────────────────────
      for (let py = wallTop; py <= wallBottom; py++) {
        // Edge darkening — pixels near top/bottom edge of wall span are darker
        const edgeDist = Math.min(py - wallTop, wallBottom - py);
        const edgeFactor = Math.min(1, edgeDist / (tunnelPxHalf * 0.15 + 1));

        // Grit texture via cheap noise
        const grit = smoothNoise1D(py * 0.23 + px * 0.07, biome.seedOffset + 99) * 0.08;

        const shadedWall: [number, number, number] = [
          Math.max(0, Math.min(255, (wallColor[0] * edgeFactor + grit * 20) | 0)),
          Math.max(0, Math.min(255, (wallColor[1] * edgeFactor + grit * 14) | 0)),
          Math.max(0, Math.min(255, (wallColor[2] * edgeFactor + grit * 10) | 0)),
        ];

        setPixel(data, px, py, W, shadedWall, 255);
      }

      // (ceilDepth used above in ceiling loop)
    }

    // ── Dust particles ────────────────────────────────────────────────────
    if (biome.dustParticles) {
      this.renderDust(data, W, H, 0, 0, state, horizonY, glowPrimRgb, fogRgb);
    }

    // ── Vignette ─────────────────────────────────────────────────────────
    this.applyVignette(data, W, H);

    // ── Blit to canvas ────────────────────────────────────────────────────
    if (this.offscreen && this.offCtx) {
      this.offCtx.putImageData(this.buf, 0, 0);
      this.ctx.save();
      this.ctx.imageSmoothingEnabled = false;
      this.ctx.drawImage(this.offscreen, 0, 0, this.canvas.width, this.canvas.height);
      this.ctx.restore();
    } else {
      // Fallback: create a temp canvas
      const tmp = document.createElement('canvas');
      tmp.width  = W;
      tmp.height = H;
      const tc = tmp.getContext('2d')!;
      tc.putImageData(this.buf, 0, 0);
      this.ctx.save();
      this.ctx.imageSmoothingEnabled = false;
      this.ctx.drawImage(tmp, 0, 0, this.canvas.width, this.canvas.height);
      this.ctx.restore();
    }

    // ── Canvas-level torch glow overlay (cheap radial gradients) ─────────
    this.renderTorchGlowOverlay(state, horizonY);
  }

  // ─── Torch management ───────────────────────────────────────────────────────
  private updateTorches(distance: number, biome: BiomeConfig, elapsed: number) {
    const { torchSpacing: spacing } = biome;

    // Determine which torch indices should be visible in [distance-2, distance+MAX_DEPTH]
    const firstIdx = Math.floor((distance - 2) / spacing);
    const lastIdx  = Math.ceil((distance + MAX_DEPTH) / spacing);

    // Rebuild torch list (small, max ~8 entries)
    this.torches = [];
    for (let i = firstIdx; i <= lastIdx && this.torches.length < MAX_TORCHES; i++) {
      const torchDist = i * spacing;
      const side = (i % 2 === 0) ? -1 : 1; // alternating sides
      let flicker = 1;
      if (biome.torchFlicker) {
        // Multi-frequency flicker
        flicker = 0.75
          + 0.15 * Math.sin(elapsed * 7.3 + i * 2.1)
          + 0.10 * Math.sin(elapsed * 17.9 + i * 5.3);
      }
      this.torches.push({ distAlongTrack: torchDist, side, flicker });
    }
  }

  private computeTorchLight(
    camDist: number,
    depth: number,
    screenXNorm: number,  // 0..1
    _elapsed: number,
    _biome: BiomeConfig,
  ): number {
    let total = 0;
    for (const t of this.torches) {
      const torchDepth = t.distAlongTrack - camDist;
      if (torchDepth < -1 || torchDepth > MAX_DEPTH) continue;

      // Depth difference between surface and torch
      const dDepth = Math.abs(depth - torchDepth);
      if (dDepth > 8) continue;

      // Torch horizontal position in screen space (0..1)
      const torchScreenX = t.side < 0 ? 0.08 : 0.92;
      const dX = Math.abs(screenXNorm - torchScreenX);

      const distFactor = 1 / (1 + torchDepth * torchDepth * 0.18 + dX * 4 + dDepth * 0.5);
      total += distFactor * t.flicker * 1.8;
    }
    return Math.min(1, total);
  }

  private renderTorchGlowOverlay(state: WorldState, horizonYRaw: number) {
    const { distance, biome } = state;
    const W = this.canvas.width;
    const H = this.canvas.height;
    // Scale horizonY to display resolution
    const horizonY = (horizonYRaw / this.rh) * H;

    for (const t of this.torches) {
      const depth = t.distAlongTrack - distance;
      if (depth <= 0.3 || depth > MAX_DEPTH * 0.7) continue;

      // Perspective-project torch position
      const perspScale = H / (depth + 0.001);
      const screenX = t.side < 0
        ? W * 0.06 + perspScale * biome.tunnelWidth * 0.25
        : W * 0.94 - perspScale * biome.tunnelWidth * 0.25;
      const screenY = horizonY - biome.tunnelHeight * perspScale * 0.3;

      const radius = Math.max(8, 60 / (depth + 0.5));
      const intensity = t.flicker * Math.min(1, 3 / (depth + 0.5));

      if (intensity < 0.02) continue;

      const grad = this.ctx.createRadialGradient(screenX, screenY, 0, screenX, screenY, radius);
      const alpha1 = Math.min(0.55, intensity * 0.6);
      const alpha2 = Math.min(0.18, intensity * 0.25);
      grad.addColorStop(0,   `rgba(220, 110, 20, ${alpha1.toFixed(3)})`);
      grad.addColorStop(0.4, `rgba(180, 70, 10, ${alpha2.toFixed(3)})`);
      grad.addColorStop(1,   'rgba(0,0,0,0)');

      this.ctx.globalCompositeOperation = 'screen';
      this.ctx.fillStyle = grad;
      this.ctx.beginPath();
      this.ctx.arc(screenX, screenY, radius, 0, Math.PI * 2);
      this.ctx.fill();
      this.ctx.globalCompositeOperation = 'source-over';

      // Tiny torch sprite
      if (depth < 14) {
        const sz = Math.max(1, 4 / depth);
        this.ctx.fillStyle = `rgba(255, 220, 80, ${Math.min(0.9, intensity)})`;
        this.ctx.fillRect(screenX - sz, screenY - sz, sz * 2, sz * 2);
      }
    }
  }

  // ─── Dust ───────────────────────────────────────────────────────────────────
  private renderDust(
    data: Uint8ClampedArray,
    W: number,
    H: number,
    _halfW: number,
    _halfH: number,
    state: WorldState,
    horizonY: number,
    glowRgb: [number, number, number],
    fogRgb: [number, number, number],
  ) {
    const dt = 1 / 60; // approximate
    const { speed } = state;

    for (let i = 0; i < this.dustCount; i++) {
      const b = i * 4;
      // Advance particle toward camera
      this.dustParticles[b + 2] -= speed * dt * 0.9;
      // Drift
      this.dustParticles[b]     += (Math.random() - 0.5) * 0.01;
      this.dustParticles[b + 1] += (Math.random() - 0.48) * 0.008; // slight upward drift

      const z = this.dustParticles[b + 2];
      if (z <= 0.2) {
        this.resetDust(i, MAX_DEPTH * 0.8 + Math.random() * MAX_DEPTH * 0.2);
        continue;
      }

      const ndcX = this.dustParticles[b];
      const ndcY = this.dustParticles[b + 1];

      // Project to screen
      const px = ((ndcX / (z * FOV_FACTOR) * 0.5 + 0.5) * W) | 0;
      const py = (horizonY - ndcY / z * (H * 0.5)) | 0;

      if (px < 0 || px >= W || py < 0 || py >= H) {
        this.resetDust(i, MAX_DEPTH * 0.5 + Math.random() * MAX_DEPTH * 0.5);
        continue;
      }

      const fog = fogFactor(z, 0.5);
      const alpha = (1 - fog) * 0.6;
      if (alpha < 0.05) continue;

      // Blend dust pixel
      const idx = (py * W + px) * 4;
      data[idx]     = Math.min(255, data[idx]     + (glowRgb[0] * 0.5 * alpha) | 0);
      data[idx + 1] = Math.min(255, data[idx + 1] + (glowRgb[1] * 0.35 * alpha) | 0);
      data[idx + 2] = Math.min(255, data[idx + 2] + (glowRgb[2] * 0.25 * alpha) | 0);
    }

    void fogRgb; // suppress lint
  }

  // ─── Vignette ────────────────────────────────────────────────────────────────
  private applyVignette(data: Uint8ClampedArray, W: number, H: number) {
    // Applied in pixel buffer at render resolution — cheap.
    const cx = W * 0.5;
    const cy = H * 0.5;
    const maxDist = Math.sqrt(cx * cx + cy * cy);

    for (let py = 0; py < H; py += 2) {  // every 2nd row for perf
      for (let px = 0; px < W; px += 2) {
        const dx = (px - cx) / cx;
        const dy = (py - cy) / cy;
        const d  = Math.sqrt(dx * dx + dy * dy);
        // Stronger vignette toward corners
        const vig = Math.pow(Math.max(0, d - 0.35), 2.2) * 1.1;
        if (vig < 0.01) continue;

        const darken = (1 - Math.min(1, vig));

        // Apply to 2x2 block
        for (let oy = 0; oy < 2 && py + oy < H; oy++) {
          for (let ox = 0; ox < 2 && px + ox < W; ox++) {
            const idx = ((py + oy) * W + (px + ox)) * 4;
            data[idx]     = (data[idx]     * darken) | 0;
            data[idx + 1] = (data[idx + 1] * darken) | 0;
            data[idx + 2] = (data[idx + 2] * darken) | 0;
          }
        }
      }
    }

    void maxDist;
  }
}

// ─── Utility ─────────────────────────────────────────────────────────────────

function fogFactor(depth: number, density: number): number {
  return 1 - Math.exp(-density * depth * 0.1);
}

function setPixel(
  data: Uint8ClampedArray,
  x: number,
  y: number,
  w: number,
  rgb: [number, number, number],
  a: number,
) {
  const i = (y * w + x) * 4;
  data[i]     = rgb[0];
  data[i + 1] = rgb[1];
  data[i + 2] = rgb[2];
  data[i + 3] = a;
}
