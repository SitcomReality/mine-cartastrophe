/**
 * WorldState — the single source of truth for the simulation.
 *
 * Kept as a plain mutable object (not React state) so the game loop
 * can mutate it every frame without triggering re-renders.
 * React only reads from it for the HUD overlay.
 */

import type { BiomeConfig } from '../biomes/types';

export interface WorldState {
  /** Distance travelled along the track (world units). Grows each frame. */
  distance: number;

  /** Current cart speed in world-units per second */
  speed: number;

  /** Target speed the cart is accelerating/decelerating toward */
  targetSpeed: number;

  /** Active biome config */
  biome: BiomeConfig;

  /** Timestamp of the previous frame, for delta-time calc */
  lastTimestamp: number;

  /** Elapsed time in seconds (unaffected by speed) */
  elapsed: number;

  /** Frames rendered (for diagnostics) */
  frameCount: number;
}

export const MIN_SPEED = 4;
export const MAX_SPEED = 28;
export const ACCELERATION = 3; // units/s²

export function createWorldState(biome: BiomeConfig): WorldState {
  return {
    distance: 0,
    speed: 8,
    targetSpeed: 8,
    biome,
    lastTimestamp: 0,
    elapsed: 0,
    frameCount: 0,
  };
}

export function tickWorld(state: WorldState, timestamp: number): number {
  const dt = state.lastTimestamp === 0
    ? 0
    : Math.min((timestamp - state.lastTimestamp) / 1000, 0.05); // cap at 50ms to avoid spiral of death

  state.lastTimestamp = timestamp;
  state.elapsed += dt;
  state.frameCount++;

  // Smoothly approach target speed
  if (state.speed < state.targetSpeed) {
    state.speed = Math.min(state.speed + ACCELERATION * dt, state.targetSpeed);
  } else if (state.speed > state.targetSpeed) {
    state.speed = Math.max(state.speed - ACCELERATION * dt, state.targetSpeed);
  }

  state.distance += state.speed * dt;

  return dt;
}
