import type { Chunk, TrackSegment } from '@/types';
import { CHUNK_LENGTH, RENDER_DISTANCE, DESPAWN_DISTANCE } from '@/constants';
import { BiomeManager } from '@/biomes/BiomeManager';
import { ObjectPool } from '@/engine/ObjectPool';

/**
 * Manages infinite chunk generation and recycling
 * Uses object pooling for performance
 */
export class ChunkManager {
  private chunks: Map<number, Chunk> = new Map();
  private biomeManager: BiomeManager;
  private chunkPool: ObjectPool<Chunk>;
  
  private nextChunkId = 0;
  private lastCameraChunk = -1;
  
  constructor(biomeManager: BiomeManager) {
    this.biomeManager = biomeManager;
    
    // Create chunk pool with factory and reset functions
    this.chunkPool = new ObjectPool<Chunk>(
      () => this.createEmptyChunk(),
      (chunk) => this.resetChunk(chunk),
      RENDER_DISTANCE + DESPAWN_DISTANCE
    );
  }
  
  private createEmptyChunk(): Chunk {
    return {
      id: 0,
      zStart: 0,
      zEnd: 0,
      track: {
        id: 0,
        position: { x: 0, y: 0, z: 0 },
        direction: 'straight',
        length: CHUNK_LENGTH,
        curvature: 0,
        pitch: 0,
      },
      objects: [],
      biomeId: 'deep_mine',
    };
  }
  
  private resetChunk(chunk: Chunk): void {
    chunk.objects.length = 0;
  }
  
  /**
   * Generate a new chunk at the given index
   */
  private generateChunk(chunkIndex: number): Chunk {
    const chunk = this.chunkPool.acquire();
    
    const zStart = chunkIndex * CHUNK_LENGTH;
    const zEnd = zStart + CHUNK_LENGTH;
    
    const biome = this.biomeManager.getBiomeAt(zStart);
    
    chunk.id = this.nextChunkId++;
    chunk.zStart = zStart;
    chunk.zEnd = zEnd;
    chunk.biomeId = biome.getId();
    
    // Generate track segment
    chunk.track = this.generateTrackSegment(chunkIndex, zStart);
    
    // Generate objects using biome
    chunk.objects = biome.generateObjects(zStart, zEnd, chunk.id);
    
    return chunk;
  }
  
  /**
   * Generate track segment for a chunk
   * Future: Will support curves and height changes
   */
  private generateTrackSegment(chunkIndex: number, zStart: number): TrackSegment {
    return {
      id: chunkIndex,
      position: { x: 0, y: 0, z: zStart },
      direction: 'straight',
      length: CHUNK_LENGTH,
      curvature: 0, // Future: Add gentle curves
      pitch: 0, // Future: Add inclines/declines
    };
  }
  
  /**
   * Update chunks based on camera position
   * Generates new chunks ahead, recycles chunks behind
   */
  update(cameraZ: number): void {
    const currentChunkIndex = Math.floor(cameraZ / CHUNK_LENGTH);
    
    // Only update if we've moved to a new chunk
    if (currentChunkIndex === this.lastCameraChunk) return;
    this.lastCameraChunk = currentChunkIndex;
    
    // Calculate range of chunks to keep
    const minChunk = currentChunkIndex - DESPAWN_DISTANCE;
    const maxChunk = currentChunkIndex + RENDER_DISTANCE;
    
    // Remove chunks that are too far behind
    for (const [index, chunk] of this.chunks.entries()) {
      const chunkIndex = Math.floor(chunk.zStart / CHUNK_LENGTH);
      if (chunkIndex < minChunk) {
        this.chunkPool.release(chunk);
        this.chunks.delete(index);
      }
    }
    
    // Generate new chunks ahead
    for (let i = minChunk; i <= maxChunk; i++) {
      if (i < 0) continue;
      if (!this.chunks.has(i)) {
        const chunk = this.generateChunk(i);
        this.chunks.set(i, chunk);
      }
    }
  }
  
  /**
   * Get all currently active chunks, sorted by z position (far to near)
   */
  getVisibleChunks(cameraZ: number): Chunk[] {
    const chunks = Array.from(this.chunks.values());
    
    // Sort far to near for proper depth rendering
    chunks.sort((a, b) => b.zStart - a.zStart);
    
    // Filter to only visible chunks
    const minZ = cameraZ - DESPAWN_DISTANCE * CHUNK_LENGTH;
    const maxZ = cameraZ + RENDER_DISTANCE * CHUNK_LENGTH;
    
    return chunks.filter(c => c.zEnd >= minZ && c.zStart <= maxZ);
  }
  
  /**
   * Get the chunk at a specific z position
   */
  getChunkAt(z: number): Chunk | undefined {
    const index = Math.floor(z / CHUNK_LENGTH);
    return this.chunks.get(index);
  }
  
  /**
   * Initialize chunks around starting position
   */
  initialize(startZ: number = 0): void {
    this.chunks.clear();
    this.lastCameraChunk = -999; // Force update
    this.update(startZ);
  }
  
  /**
   * Get stats for debugging
   */
  getStats(): { activeChunks: number; poolSize: number } {
    return {
      activeChunks: this.chunks.size,
      poolSize: this.chunkPool.size(),
    };
  }
}
