import type { BiomeConfig, BiomeTransition } from '@/types';
import { BaseBiome } from './BaseBiome';
import { DeepMineBiome } from './DeepMineBiome';
import { CavernousAbyssBiome } from './CavernousAbyssBiome';
import { WORLD_SEED, CHUNK_LENGTH } from '@/constants';
import { color, clamp } from '@/utils/math';

// Biome sequence configuration
interface BiomeZone {
  biomeId: string;
  startZ: number;
  endZ: number;
}

/**
 * Manages biome selection and smooth transitions between biomes
 */
export class BiomeManager {
  private biomes: Map<string, BaseBiome> = new Map();
  private zones: BiomeZone[] = [];
  private transitionLength = CHUNK_LENGTH * 3; // 3 chunks to transition
  
  constructor(seed: number = WORLD_SEED) {
    // Register available biomes
    this.registerBiome(new DeepMineBiome(seed));
    this.registerBiome(new CavernousAbyssBiome(seed + 1));
    
    // Define the biome sequence (can be procedurally generated later)
    this.generateBiomeZones();
  }
  
  private registerBiome(biome: BaseBiome): void {
    this.biomes.set(biome.getId(), biome);
  }
  
  /**
   * Generate the sequence of biome zones
   * Future: This can be procedurally generated based on distance
   */
  private generateBiomeZones(): void {
    // For now, a simple repeating pattern with transitions
    // Start with deep mine, then alternate
    const zoneLength = CHUNK_LENGTH * 15; // ~15 chunks per zone
    
    this.zones = [
      { biomeId: 'deep_mine', startZ: 0, endZ: zoneLength },
      { biomeId: 'cavernous_abyss', startZ: zoneLength, endZ: zoneLength * 2 },
      { biomeId: 'deep_mine', startZ: zoneLength * 2, endZ: zoneLength * 3 },
      { biomeId: 'cavernous_abyss', startZ: zoneLength * 3, endZ: zoneLength * 4 },
      // Pattern continues...
    ];
  }
  
  /**
   * Get the biome zone for a given z position
   */
  private getZoneAt(z: number): BiomeZone {
    // Find the zone, or extrapolate if beyond defined zones
    for (const zone of this.zones) {
      if (z >= zone.startZ && z < zone.endZ) {
        return zone;
      }
    }
    
    // If beyond defined zones, calculate based on repeating pattern
    const totalLength = this.zones[this.zones.length - 1]?.endZ || CHUNK_LENGTH * 30;
    const cycleZ = z % totalLength;
    
    for (const zone of this.zones) {
      if (cycleZ >= zone.startZ && cycleZ < zone.endZ) {
        return zone;
      }
    }
    
    // Default fallback
    return this.zones[0];
  }
  
  /**
   * Get the appropriate biome for a given z position
   */
  getBiomeAt(z: number): BaseBiome {
    const zone = this.getZoneAt(z);
    return this.biomes.get(zone.biomeId) || this.biomes.get('deep_mine')!;
  }
  
  /**
   * Get current biome configuration at position
   */
  getCurrentConfig(z: number): BiomeConfig {
    const transition = this.getTransitionBlend(z);
    
    if (transition) {
      return this.blendConfigs(
        transition.fromBiome,
        transition.toBiome,
        transition.progress
      );
    }
    
    return this.getBiomeAt(z).getConfig();
  }
  
  /**
   * Get a specific biome by ID
   */
  getBiome(id: string): BaseBiome | undefined {
    return this.biomes.get(id);
  }
  
  /**
   * Get all registered biome IDs
   */
  getBiomeIds(): string[] {
    return Array.from(this.biomes.keys());
  }
  
  /**
   * Calculate transition blend between two biomes
   * Returns null if not in a transition zone
   */
  getTransitionBlend(z: number): BiomeTransition | null {
    const currentZone = this.getZoneAt(z);
    const nextZoneStartZ = currentZone.endZ;
    const transitionStartZ = nextZoneStartZ - this.transitionLength;
    
    // Check if we're in the transition zone at the end of current biome
    if (z >= transitionStartZ && z < nextZoneStartZ) {
      const nextZone = this.getZoneAt(nextZoneStartZ);
      
      // Don't transition to the same biome
      if (nextZone.biomeId === currentZone.biomeId) {
        return null;
      }
      
      const fromBiome = this.biomes.get(currentZone.biomeId)!;
      const toBiome = this.biomes.get(nextZone.biomeId)!;
      
      // Calculate smooth transition progress using smoothstep
      const rawProgress = (z - transitionStartZ) / this.transitionLength;
      const progress = this.smoothstep(rawProgress);
      
      return {
        fromBiome: fromBiome.getConfig(),
        toBiome: toBiome.getConfig(),
        progress,
      };
    }
    
    return null;
  }
  
  /**
   * Smoothstep interpolation for smoother transitions
   */
  private smoothstep(t: number): number {
    const x = clamp(t, 0, 1);
    return x * x * (3 - 2 * x);
  }
  
  /**
   * Blend two biome configurations together
   */
  private blendConfigs(from: BiomeConfig, to: BiomeConfig, t: number): BiomeConfig {
    return {
      id: t < 0.5 ? from.id : to.id,
      name: t < 0.5 ? from.name : to.name,
      
      // Blend colors
      fogColor: color.lerp(from.fogColor, to.fogColor, t),
      fogDensity: this.lerp(from.fogDensity, to.fogDensity, t),
      ambientLight: this.lerp(from.ambientLight, to.ambientLight, t),
      
      // Blend dimensions (this creates the opening/closing effect!)
      tunnelWidth: this.lerp(from.tunnelWidth, to.tunnelWidth, t),
      tunnelHeight: this.lerp(from.tunnelHeight, to.tunnelHeight, t),
      wallColor: color.lerp(from.wallColor, to.wallColor, t),
      floorColor: color.lerp(from.floorColor, to.floorColor, t),
      ceilingColor: color.lerp(from.ceilingColor, to.ceilingColor, t),
      
      // Blend track colors
      trackColor: color.lerp(from.trackColor, to.trackColor, t),
      railColor: color.lerp(from.railColor, to.railColor, t),
      
      // Use weighted average for generation params
      supportBeamFrequency: this.lerp(from.supportBeamFrequency, to.supportBeamFrequency, t),
      lanternFrequency: this.lerp(from.lanternFrequency, to.lanternFrequency, t),
      decorationDensity: this.lerp(from.decorationDensity, to.decorationDensity, t),
      
      // Take object weights from dominant biome
      objectWeights: t < 0.5 ? from.objectWeights : to.objectWeights,
      
      // Blend wall presence using smooth threshold (walls fade out around 0.3-0.5 mark)
      // This creates a gradual "opening" effect as walls recede
      hasLeftWall: t < 0.4 ? from.hasLeftWall : to.hasLeftWall,
      hasRightWall: t < 0.4 ? from.hasRightWall : to.hasRightWall,
      hasCeiling: t < 0.6 ? from.hasCeiling : to.hasCeiling, // Ceiling transitions later
      hasFloor: t < 0.35 ? from.hasFloor : to.hasFloor, // Floor transitions early (dramatic!)
      
      // Blend floor/ledge width smoothly
      floorEdgeWidth: this.lerp(from.floorEdgeWidth, to.floorEdgeWidth, t),
      
      // Blend abyss properties
      abyssDepth: this.lerp(from.abyssDepth, to.abyssDepth, t),
      abyssGlowColor: from.abyssGlowColor && to.abyssGlowColor
        ? color.lerp(from.abyssGlowColor, to.abyssGlowColor, t)
        : (t < 0.5 ? from.abyssGlowColor : to.abyssGlowColor),
      abyssGlowIntensity: this.lerp(from.abyssGlowIntensity, to.abyssGlowIntensity, t),
    };
  }
  
  private lerp(a: number, b: number, t: number): number {
    return a + (b - a) * t;
  }
  
  /**
   * Get the transition length in world units
   */
  getTransitionLength(): number {
    return this.transitionLength;
  }
  
  /**
   * Check if position is in a transition zone
   */
  isInTransition(z: number): boolean {
    return this.getTransitionBlend(z) !== null;
  }
}
