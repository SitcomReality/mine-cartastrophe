// Core geometry types
export interface Vec3 {
  x: number;
  y: number;
  z: number;
}

export interface Vec2 {
  x: number;
  y: number;
}

// Color with alpha support
export interface Color {
  r: number;
  g: number;
  b: number;
  a?: number;
}

// Track segment types
export type TrackDirection = 'straight' | 'left' | 'right' | 'up' | 'down';

export interface TrackSegment {
  id: number;
  position: Vec3;
  direction: TrackDirection;
  length: number;
  curvature: number; // 0 = straight, negative = left, positive = right
  pitch: number; // vertical angle
}

// Environmental objects in the mine
export interface MineObject {
  id: number;
  type: MineObjectType;
  position: Vec3;
  rotation: number;
  scale: number;
}

export type MineObjectType = 
  | 'support_beam'
  | 'lantern'
  | 'stalactite'
  | 'stalagmite'
  | 'rubble'
  | 'rail_tie'
  | 'crystal'
  | 'chain'
  | 'crate'
  | 'barrel';

// Chunk system for infinite world
export interface Chunk {
  id: number;
  zStart: number;
  zEnd: number;
  track: TrackSegment;
  objects: MineObject[];
  biomeId: string;
}

// Biome configuration
export interface BiomeConfig {
  id: string;
  name: string;
  
  // Visual atmosphere
  fogColor: Color;
  fogDensity: number;
  ambientLight: number;
  
  // Tunnel properties
  tunnelWidth: number;
  tunnelHeight: number;
  wallColor: Color;
  floorColor: Color;
  ceilingColor: Color;
  
  // Track properties
  trackColor: Color;
  railColor: Color;
  
  // Generation parameters
  supportBeamFrequency: number;
  lanternFrequency: number;
  decorationDensity: number;
  
  // Object spawn weights
  objectWeights: Partial<Record<MineObjectType, number>>;
  
  // Abyss/openness properties
  hasLeftWall: boolean;
  hasRightWall: boolean;
  hasCeiling: boolean;
  hasFloor: boolean; // If false, track is on a bridge/ledge
  floorEdgeWidth: number; // Width of the floor/ledge (when hasFloor is false for open pits)
  abyssDepth: number; // Visual depth of the void (for glow effects)
  abyssGlowColor?: Color; // Optional glow from the depths
  abyssGlowIntensity: number; // 0-1
}

// Transition state between biomes
export interface BiomeTransition {
  fromBiome: BiomeConfig;
  toBiome: BiomeConfig;
  progress: number; // 0-1, where 0 is fully "from" and 1 is fully "to"
}

// Extended chunk with transition support
export interface ChunkTransitionData {
  transition?: BiomeTransition;
  ledgeSide?: 'left' | 'right' | 'both' | 'none'; // Which side has the abyss
}

// Camera state
export interface CameraState {
  position: Vec3;
  rotation: Vec3; // pitch, yaw, roll
  fov: number;
  shake: number;
}

// Render state for performance tracking
export interface RenderStats {
  fps: number;
  drawCalls: number;
  visibleChunks: number;
  objectsRendered: number;
}

// Game state
export interface GameState {
  speed: number;
  distance: number;
  time: number;
  paused: boolean;
  currentBiome: string;
}
